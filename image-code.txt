window.TEKZAA_RENDER = function(time, ctx, width, height) {
    function safeNumber(v, fallback) {
        return Number.isFinite(v) ? v : fallback;
    }
function safePositive(v, fallback) {
    return Number.isFinite(v) && v > 0 ? v : fallback;
}

function safeClamp(v, min, max, fallback) {
    if (!Number.isFinite(v)) return fallback;
    return Math.max(min, Math.min(max, v));
}

function safeLerp(a, b, t) {
    a = safeNumber(a, 0);
    b = safeNumber(b, 0);
    t = safeClamp(t, 0, 1, 0);
    return a + (b - a) * t;
}

function safeGradient(c, x0, y0, x1, y1) {
    return c.createLinearGradient(
        safeNumber(x0, 0),
        safeNumber(y0, 0),
        safeNumber(x1, 1),
        safeNumber(y1, 1)
    );
}

function safeRadialGradient(c, x0, y0, r0, x1, y1, r1) {
    return c.createRadialGradient(
        safeNumber(x0, 0),
        safeNumber(y0, 0),
        Math.max(0, safeNumber(r0, 0)),
        safeNumber(x1, 0),
        safeNumber(y1, 0),
        Math.max(0.001, safeNumber(r1, 1))
    );
}

function progressAt(t, start, duration) {
    t = safeNumber(t, 0);
    start = safeNumber(start, 0);
    duration = safePositive(duration, 0.001);
    return safeClamp((t - start) / duration, 0, 1, 0);
}

function easeOutCubic(t) {
    t = safeClamp(t, 0, 1, 0);
    return 1 - Math.pow(1 - t, 3);
}

function easeOutBack(t) {
    t = safeClamp(t, 0, 1, 0);
    var c1 = 1.70158;
    var c3 = c1 + 1;
    return 1 + c3 * Math.pow(t - 1, 3) + c1 * Math.pow(t - 1, 2);
}

function drawText(c, text, x, y, size, weight, color, align, alpha) {
    x = safeNumber(x, 0);
    y = safeNumber(y, 0);
    size = safePositive(size, 12);
    alpha = safeClamp(alpha, 0, 1, 1);
    c.save();
    c.globalAlpha = alpha;
    c.font = String(weight || "400") + " " + size + "px Arial, Helvetica, sans-serif";
    c.fillStyle = color || "#333333";
    c.textAlign = align || "left";
    c.textBaseline = "middle";
    c.fillText(String(text || ""), x, y);
    c.restore();
}

function roundRect(c, x, y, w, h, r) {
    x = safeNumber(x, 0);
    y = safeNumber(y, 0);
    w = safePositive(w, 1);
    h = safePositive(h, 1);
    r = safeClamp(r, 0, Math.min(w, h) * 0.5, 0);
    c.beginPath();
    c.moveTo(x + r, y);
    c.lineTo(x + w - r, y);
    c.quadraticCurveTo(x + w, y, x + w, y + r);
    c.lineTo(x + w, y + h - r);
    c.quadraticCurveTo(x + w, y + h, x + w - r, y + h);
    c.lineTo(x + r, y + h);
    c.quadraticCurveTo(x, y + h, x, y + h - r);
    c.lineTo(x, y + r);
    c.quadraticCurveTo(x, y, x + r, y);
    c.closePath();
}

    sectorAngles.push({
        start: runningAngle,
        end: runningAngle + sweep,
        mid: runningAngle + sweep * 0.5
    });
    runningAngle += sweep;

for (var si = values.length - 1; si >= 0; si--) {
    var sec = sectorAngles[si];
    var a0 = sec.start;
    var a1 = sec.end;

    var steps = 20;
    var isBottomVisible = Math.sin(sec.mid) > -0.25;
    var drawDepth = isBottomVisible ? currentDepth : currentDepth * 0.72;

    for (var layer = Math.max(1, Math.floor(drawDepth)); layer >= 1; layer--) {
        var yy = layer;

        ctx.beginPath();
        ctx.moveTo(
            Math.cos(a0) * currentRx,
            Math.sin(a0) * currentRy + yy
        );

        for (var st = 0; st <= steps; st++) {
            var u = st / steps;
            var ang = safeLerp(a0, a1, u);
            ctx.lineTo(
                Math.cos(ang) * currentRx,
                Math.sin(ang) * currentRy + yy
            );
        }

        ctx.lineTo(
            Math.cos(a0) * currentRx,
            Math.sin(a0) * currentRy + yy
        );
        ctx.closePath();

        var sideGrad = safeGradient(
            ctx,
            -currentRx,
            -currentRy + yy,
            currentRx,
            currentRy + yy
        );
        sideGrad.addColorStop(0, colors[si][1]);
        sideGrad.addColorStop(0.5, colors[si][0]);
        sideGrad.addColorStop(1, colors[si][1]);

        ctx.fillStyle = sideGrad;
        ctx.fill();
    }
}

    ctx.moveTo(0, 0);

}

    return {
        x: cx + Math.cos(angle) * distance,
        y: cy + Math.sin(angle) * distance * (safePositive(ry, 1) / safePositive(rx, 1))
    };
}
